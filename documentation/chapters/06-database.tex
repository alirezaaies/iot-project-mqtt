\begin{phasebox}
\textbf{هدف:} ساخت پایگاه داده و ذخیره پایدار یک داده سنسور.\\
\textbf{پیش‌نیاز:} قالب پیام داده مشخص باشد.\\
\textbf{معیار پایان:} یک داده ذخیره و پس از اجرای دوباره برنامه خوانده شود.
\end{phasebox}

\section{محل کد}
ساخت جدول‌ها، راه‌اندازی پایگاه داده و تمام توابع خواندن و نوشتن در پوشه
\lr{\texttt{code/database}}
قرار می‌گیرند. فایل واقعی پایگاه داده در پوشه
\lr{\texttt{runtime}}
ساخته می‌شود و وارد
\lr{Git}
نخواهد شد.

\section{مدل اولیه داده}
در ابتدا فقط اطلاعات دستگاه، سنسور، داده اندازه‌گیری‌شده و فرمان ذخیره می‌شوند. جدول جدید فقط زمانی ساخته می‌شود که یک نیاز واقعی وجود داشته باشد.

\section{قواعد ساده}
شناسه دستگاه، زمان، واحد و مقدار باید روشن باشند. زمان در پایگاه داده با معیار جهانی ذخیره می‌شود. دستورهای مستقیم
\lr{SQL}
نباید در پوشه بک‌اند یا برنامه وب پخش شوند.

\section{آزمایش مستقل}
\begin{itemize}
    \item ساخت یک فایل موقت
    \lr{SQLite}
    \item درج یک داده آزمایشی؛
    \item بستن و بازکردن برنامه؛
    \item خواندن همان داده و مقایسه مقدارها.
\end{itemize}

\section{تغییر پایگاه داده در آینده}
اگر پایگاه داده تغییر کند، توابع عمومی خواندن و نوشتن ثابت می‌مانند و پیاده‌سازی داخل پوشه پایگاه داده عوض می‌شود.
