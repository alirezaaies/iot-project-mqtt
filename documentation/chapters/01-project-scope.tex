\begin{phasebox}
\textbf{هدف:} تعیین کوچک‌ترین نسخه قابل استفاده و مرز پروژه.\\
\textbf{پیش‌نیاز:} ندارد.\\
\textbf{معیار پایان:} یک سناریوی کامل با ورودی، خروجی و معیار پذیرش مشخص باشد.
\end{phasebox}

\section{مسئله}
سامانه باید داده یک سنسور را از یک دستگاه دریافت و روی دیسک ذخیره کند، آخرین داده‌ها و وضعیت دستگاه را در وب نمایش دهد و یک فرمان ساده را به دستگاه ارسال کند.

\section{نسخه اولیه}
نسخه اولیه فقط شامل موارد زیر است:

\begin{itemize}
    \item یک برد
    \lr{ESP8266}
    و یک سنسور یا مقدار شبیه‌سازی‌شده؛
    \item یک کارگزار محلی
    \lr{MQTT}
    برای جابه‌جایی پیام‌ها؛
    \item پایگاه داده
    \lr{SQLite}
    برای ذخیره اطلاعات؛
    \item یک برنامه ساده
    \lr{Python}
    برای دریافت پیام‌ها؛
    \item یک برنامه وب ساده با
    \lr{Flask}
    برای نمایش اطلاعات.
\end{itemize}

\section{خارج از محدوده نسخه اولیه}
اپلیکیشن موبایل، تحلیل پیشرفته، مقیاس ابری، به‌روزرسانی بی‌سیم دستگاه و پشتیبانی هم‌زمان از چند پایگاه داده در نسخه اولیه پیاده‌سازی نمی‌شوند. فقط ساختار پوشه‌ها به شکلی نگه داشته می‌شود که توسعه بعدی دشوار نباشد.

\section{معیارهای پذیرش}
\begin{itemize}
    \item داده معتبر پس از توقف و اجرای دوباره برنامه باقی بماند.
    \item آخرین وضعیت دستگاه و داده‌های اخیر در وب دیده شود.
    \item یک فرمان ساده از وب ارسال و نتیجه آن از دستگاه دریافت شود.
    \item هر قسمت قبل از اتصال به قسمت بعد، یک تست مستقل داشته باشد.
\end{itemize}
