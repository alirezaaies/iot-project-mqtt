\begin{phasebox}
\textbf{هدف:} خواندن اطلاعات ذخیره‌شده از طریق برنامه وب.\\
\textbf{پیش‌نیاز:} داده آزمایشی در پایگاه داده وجود داشته باشد.\\
\textbf{معیار پایان:} فهرست دستگاه‌ها و داده‌های اخیر در پاسخ وب دریافت شوند.
\end{phasebox}

\section{محل کد}
برنامه
\lr{Flask}
مسیرهای وب و فایل‌های مربوط به صفحه‌ها در پوشه
\lr{\texttt{code/webapp}}
قرار می‌گیرند.

\section{تقسیم مسئولیت}
برنامه وب ورودی کاربر را می‌گیرد و تابع مناسب بک‌اند را فراخوانی می‌کند. ارتباط مستقیم با برد و دستورهای مستقیم پایگاه داده در مسیرهای وب نوشته نمی‌شوند.

\section{خروجی‌های اولیه}
در مرحله اول فقط این خروجی‌ها لازم هستند:

\begin{itemize}
    \item فهرست دستگاه‌ها؛
    \item جزئیات یک دستگاه؛
    \item داده‌های اخیر سنسور؛
    \item پیام خطای قابل فهم.
\end{itemize}

\section{آزمایش مستقل}
پاسخ‌های
\lr{HTTP}
با داده آزمایشی بررسی می‌شوند. حالت دستگاه موجود، دستگاه ناموجود و نبود داده باید نتیجه مشخص داشته باشد.

\section{تغییر ابزار وب}
اگر در آینده
\lr{Django}
جایگزین
\lr{Flask}
شود، پوشه برنامه وب بازنویسی می‌شود؛ کد برد، بک‌اند و پایگاه داده نباید به دلیل این تغییر بازنویسی شوند.
